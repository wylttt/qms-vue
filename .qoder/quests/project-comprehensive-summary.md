// JSON驱动的动态评分算法
calculateScore(templateContent, answerContent):

            option = findOption(templateContent, optionId)
            totalScore += option.score
    RETURN totalScore

// 重点人群自动判定
isFocusGroup = totalScore >= 60 ? 1 : 0
