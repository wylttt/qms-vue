ALTER TABLE gc_sampling_site ADD COLUMN daily_capacity INT DEFAULT 100 COMMENT '每日预约容量';

// 预约检测逻辑
public boolean checkCapacity(Long siteId, LocalDate appointmentDate) {
    // 查询当日预约数
    int count = appointmentMapper.countByDateAndSite(appointmentDate, siteId);
    // 查询采血点容量
    int capacity = siteMapper.selectById(siteId).getDailyCapacity();
    return count < capacity;
}
